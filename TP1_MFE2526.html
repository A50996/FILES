$\frac{d\vec X}{dt}=\mathbf{G} (x,y) = \begin{cases}\frac{dx}{dt}=\alpha x - \beta xy\\ \\ \frac{dy}{dt}=\delta xy - \gamma y \end{cases}$

Ponto fixo: $\mathbf{G}=0$¶

$\Downarrow$

$\mathbf{G}= 0 \Rightarrow \begin{cases}\alpha x - \beta xy = 0\\ \\ \delta xy - \gamma y = 0 \end{cases} \quad \Leftrightarrow \quad \mathbf{G^*}_1 = (0,0) \;\; ; \;\; \begin{cases}\alpha - \beta y = 0\\ \\ \delta x - \gamma = 0 \end{cases} \quad \Leftrightarrow \quad \mathbf{G^*}_1 = (0,0) \;\; ; \;\; \mathbf{G^*}_2 = (\frac{\alpha}{\beta},\frac{\gamma}{\delta})$

import numpy as np

class LotkaVolterra:
    def __init__(self, X0, k, metodo='rk4'):
        self.X = np.array(X0, dtype=float)
        self.k = k
        self.metodo = metodo

    def G(self, X):
        x, y = X
        return np.array([x * (1 - y),
                         self.k * y * (x - 1)])

    def H(self):
        x, y = self.X
        return x - np.log(x) + self.k * (y - np.log(y))

    def passo(self, dt):
        if self.metodo == 'euler':
            self._euler(dt)
        elif self.metodo == 'rk4':
            self._rk4(dt)

    def _euler(self, dt):
        self.X += dt * self.G(self.X)

    def _rk4(self, dt):
        k1 = self.G(self.X)
        k2 = self.G(self.X + dt/2 * k1)
        k3 = self.G(self.X + dt/2 * k2)
        k4 = self.G(self.X + dt   * k3)
        self.X += dt/6 * (k1 + 2*k2 + 2*k3 + k4)
